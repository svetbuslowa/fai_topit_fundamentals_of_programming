**Код:**

```python
def classify_student(score: int, work_type: str, absences: int) -> str:
    # Часть 1. Проверка диапазонов
    if not (0 <= score <= 100):
        return "Ошибка: результат должен быть от 0 до 100."
    if not (0 <= absences <= 5):
        return "Ошибка: количество пропусков должно быть от 0 до 5."
    
    # Часть 2. Определение уровня
    if score >= 90: level = "отличный"
    elif score >= 75: level = "хороший"
    elif score >= 60: level = "удовлетворительный"
    else: level = "недостаточный"
    
    levels = ["недостаточный", "удовлетворительный", "хороший", "отличный"]
    if absences > 3:
        idx = levels.index(level)
        level = levels[max(0, idx - 1)]
        
    # Часть 3. Сопоставление с образцом
    match work_type:
        case "exam": type_desc = "экзаменационная работа"
        case "project": type_desc = "проектная работа"
        case "practice": type_desc = "практическая работа"
        case _: type_desc = "неизвестный тип работы"
        
    return f"Уровень: {level}, Тип: {type_desc}"

# Тестовые сценарии
test_cases = [
    (0, "exam", 0), (59, "project", 2), (60, "practice", 3), (74, "exam", 4),
    (75, "project", 0), (89, "practice", 5), (90, "exam", 3), (100, "project", 4)
]
for s, w, a in test_cases:
    print(f"({s}, {w}, {a}) -> {classify_student(s, w, a)}")
```
**Ожидаемый результат:**
Программа должна корректно обрабатывать границы диапазонов. При пропусках > 3 уровень должен снижаться на ступень, но не ниже "недостаточный»". match/case должен корректно отрабатывать для известных и неизвестных типов.
